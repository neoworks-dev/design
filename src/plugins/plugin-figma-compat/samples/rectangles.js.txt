// Sample Figma plugin 1 (from Figma's plugin quickstart): a row of orange rectangles, selected and
// scrolled into view. Runs unchanged.
const nodes = [];
for (let i = 0; i < 5; i++) {
	const rect = figma.createRectangle();
	rect.x = i * 150;
	rect.fills = [{ type: 'SOLID', color: { r: 1, g: 0.5, b: 0 } }];
	figma.currentPage.appendChild(rect);
	nodes.push(rect);
}
figma.currentPage.selection = nodes;
figma.viewport.scrollAndZoomIntoView(nodes);
figma.closePlugin();
